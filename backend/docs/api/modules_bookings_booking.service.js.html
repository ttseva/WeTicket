/**
 * Сервис работы с бронированиями.
 */
class BookingService {
  /**
   * Создаёт черновик бронирования в статусе `pending`.
   *
   * @param {object} payload Входные данные для бронирования.
   * @param {string} payload.userId Идентификатор пользователя.
   * @param {string} payload.eventId Идентификатор мероприятия.
   * @param {string[]} payload.seatIds Массив идентификаторов мест.
   * @returns {{bookingId: string, status: string, expiresAt: string}} Результат создания брони.
   * @throws {Error} Бросает ошибку, если не переданы обязательные поля.
   *
   * @example
   * const service = new BookingService();
   * const booking = service.createPendingBooking({
   *   userId: "u-1",
   *   eventId: "e-10",
   *   seatIds: ["s-1", "s-2"],
   * });
   */
  createPendingBooking(payload) {
    if (!payload?.userId || !payload?.eventId || !Array.isArray(payload?.seatIds)) {
      throw new Error("Некорректные данные для создания бронирования");
    }

    const expiresAt = new Date(Date.now() + 15 * 60 * 1000).toISOString();

    return {
      bookingId: `booking_${Date.now()}`,
      status: "pending",
      expiresAt,
    };
  }

  /**
   * Переводит просроченное бронирование в статус `expired`.
   *
   * @param {string} bookingId Идентификатор бронирования.
   * @returns {{bookingId: string, status: string}} Обновлённое состояние бронирования.
   */
  expireBooking(bookingId) {
    return {
      bookingId,
      status: "expired",
    };
  }
}

module.exports = { BookingService };
