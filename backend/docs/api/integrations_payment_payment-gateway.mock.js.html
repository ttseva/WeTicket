/**
 * mock-класс платёжного шлюза.
 * Нужен для разработки и тестов до подключения реального провайдера.
 */
class PaymentGatewayMock {
  /**
   * Имитирует проведение платежа.
   *
   * @param {object} input Параметры платежа.
   * @param {string} input.bookingId Идентификатор бронирования.
   * @param {number} input.amount Сумма к оплате.
   * @param {"card"|"wallet"} input.method Метод оплаты.
   * @returns {{transactionId: string, status: "success"|"failed"}} Результат транзакции.
   */
  processPayment(input) {
    const hasValidAmount = Number(input?.amount) > 0;

    if (!hasValidAmount) {
      return {
        transactionId: `tx_${Date.now()}`,
        status: "failed",
      };
    }

    return {
      transactionId: `tx_${Date.now()}`,
      status: "success",
    };
  }

  /**
   * Имитирует возврат средств по транзакции.
   *
   * @param {string} transactionId Идентификатор транзакции.
   * @returns {{transactionId: string, status: "refunded"}} Результат возврата.
   */
  refundPayment(transactionId) {
    return {
      transactionId,
      status: "refunded",
    };
  }
}

module.exports = { PaymentGatewayMock };
