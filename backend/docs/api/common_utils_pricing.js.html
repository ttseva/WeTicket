/**
 * Рассчитывает итоговую сумму бронирования по выбранным местам.
 *
 * @param {Array<{seatId: string, price: number}>} seats Массив мест с ценой.
 * @param {number} [serviceFeePercent=0] Процент сервисного сбора.
 * @returns {{subtotal: number, fee: number, total: number}} Разбивка итоговой стоимости.
 *
 * @example
 * const result = calculateBookingTotal(
 *   [{ seatId: "s-1", price: 1500 }, { seatId: "s-2", price: 2000 }],
 *   5
 * );
 * // => { subtotal: 3500, fee: 175, total: 3675 }
 */
function calculateBookingTotal(seats, serviceFeePercent = 0) {
  const subtotal = seats.reduce((sum, seat) => sum + Number(seat.price || 0), 0);
  const fee = Math.round((subtotal * serviceFeePercent) / 100);
  const total = subtotal + fee;

  return { subtotal, fee, total };
}

module.exports = { calculateBookingTotal };
