/**
 * Middleware проверки ролей пользователя.
 * Возвращает 403, если роль пользователя отсутствует в списке разрешённых.
 *
 * @param {...string} allowedRoles Допустимые роли (например: `admin`, `organizer`).
 * @returns {Function} Express middleware.
 *
 * @example
 * router.get("/admin/statistics", authorizeRoles("admin"), handler);
 */
function authorizeRoles(...allowedRoles) {
  return (req, res, next) => {
    const userRole = req.user?.role;

    if (!userRole || !allowedRoles.includes(userRole)) {
      return res.status(403).json({
        error: "FORBIDDEN",
        message: "Недостаточно прав для выполнения операции",
      });
    }

    return next();
  };
}

module.exports = { authorizeRoles };
